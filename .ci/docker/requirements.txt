# APIs the codebase uses (e.g. torch.distributed.checkpoint.HuggingFaceStorageWriter
# only landed in nightlies). Install with:
#   pip install --pre --upgrade torch torchvision torchaudio \
#       --index-url https://download.pytorch.org/whl/nightly/cu124

# path; nightly is strongly preferred.
torch >= 2.6
torchdata >= 0.8.0
datasets >= 3.6.0, < 4.8.0
tensorboard
wandb
fsspec
tyro >= 1.0.5

safetensors
einops
